Usage dalam Admin Service:
// src/admin/tenant-service.ts
import { publicPool } from './connections.js';
import { tenantManager } from '../multi-tenant/connection-manager.js';
export class TenantService {
  // CREATE tenant (insert ke public.tenants)
  async createTenant(input: CreateTenantInput) {
    // 1. Insert to public.tenants
    const result = await publicPool.query(`
      INSERT INTO public.tenants (id, name, slug, schema_name, status)
      VALUES ($1, $2, $3, $4, 'active')
      RETURNING *
    `, [input.id, input.name, input.slug, input.schemaName]);
    
    // 2. Provision schema (run script)
    await this.provisionSchema(result.rows[0]);
    
    // 3. Reload tenant registry
    await tenantManager.reload();
    
    return result.rows[0];
  }
  
  // LIST tenants (read from public.tenants)
  async listTenants() {
    const result = await publicPool.query(`
      SELECT * FROM public.tenants ORDER BY created_at DESC
    `);
    return result.rows;
  }
  
  // UPDATE tenant (modify public.tenants)
  async suspendTenant(tenantId: string) {
    await publicPool.query(`
      UPDATE public.tenants 
      SET status = 'suspended', updated_at = NOW()
      WHERE id = $1
    `, [tenantId]);
    
    await tenantManager.reload();
  }
  
  // Get tenant metrics (query tenant schema via manager)
  async getTenantMetrics(tenantId: string) {
    const client = tenantManager.getClient(tenantId);
    
    const userCount = await client.user.count();
    const sessionCount = await client.session.count({
      where: { expiresAt: { gt: new Date() } }
    });
    
    return { userCount, sessionCount };
  }
}
